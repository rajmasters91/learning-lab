start=$(date +%s.%N)
curl -sN http://localhost:8000/v1/chat/completions \
  -H "Content-Type: application/json" \
  -d @long_prompt.json \
| while IFS= read -r line; do
    [ -z "$line" ] && continue
    now=$(date +%s.%N)
    printf '%8.3f  %s\n' "$(echo "$now - $start" | bc)" "${line:0:80}"
  done
